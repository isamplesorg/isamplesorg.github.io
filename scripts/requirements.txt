# Dependencies for one-shot scripts in this directory (NOT the Quarto site

# scripts/setup_pipeline_venv.sh). Install into an isolated venv instead:

#   uv venv scripts/.venv && uv pip install -p scripts/.venv -r scripts/requirements.txt
